การเชื่อมต่อ Google Sheets กลาง

ให้อาจารย์กรอกแล้วข้อมูลถูกส่งไปบันทึกลงใน Google Sheet อัตโนมัติ

* หากยังไม่มี URL สามารถบันทึกในเบราว์เซอร์และกด "ดาวน์โหลด Sheet (.xlsx)" ได้ทันที

โค้ด Google Apps Script (doPost) สำหรับวางใน Sheet: 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(),
    data.executionStatus, // สำเร็จแล้ว หรือ กำลังดำเนินการ
    data.workYear,
    data.workCategory,
    data.workTitle,
    data.facultyName,
    data.department,
    data.authorRole,
    data.coAuthors,
    data.detail1, // Journal / CreativeType / IPType
    data.detail2, // Database-QualityLevel / Utilization / IPStatus
    data.grantType,
    data.grantSource,
    data.budget,
    data.inProgressInfo || '-',
    data.link
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
